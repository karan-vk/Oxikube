# Mimir (mimir): rate window 2m
# one line per target and metric; `-` is a combination with no query

## cluster
cpu_usage: sum(rate(node_cpu_seconds_total{mode!="idle"}[2m]))
cpu_requests: sum(kube_pod_container_resource_requests{resource="cpu"} * on (namespace, pod) group_left() (kube_pod_status_phase{phase=~"Pending|Running"} == 1))
cpu_limits: sum(kube_pod_container_resource_limits{resource="cpu"} * on (namespace, pod) group_left() (kube_pod_status_phase{phase=~"Pending|Running"} == 1))
memory_usage: sum(node_memory_MemTotal_bytes - node_memory_MemAvailable_bytes)
memory_requests: sum(kube_pod_container_resource_requests{resource="memory"} * on (namespace, pod) group_left() (kube_pod_status_phase{phase=~"Pending|Running"} == 1))
memory_limits: sum(kube_pod_container_resource_limits{resource="memory"} * on (namespace, pod) group_left() (kube_pod_status_phase{phase=~"Pending|Running"} == 1))
network_rx: sum(rate(node_network_receive_bytes_total{device!~"lo|veth.*|docker.*|br-.*|cni.*|flannel.*|cali.*|cilium.*|lxc.*|tunl.*|vxlan.*|kube-.*|virbr.*|weave|tap.*"}[2m]))
network_tx: sum(rate(node_network_transmit_bytes_total{device!~"lo|veth.*|docker.*|br-.*|cni.*|flannel.*|cali.*|cilium.*|lxc.*|tunl.*|vxlan.*|kube-.*|virbr.*|weave|tap.*"}[2m]))
disk_usage: sum(node_filesystem_size_bytes{mountpoint="/",fstype!~"tmpfs|ramfs|squashfs"} - node_filesystem_avail_bytes{mountpoint="/",fstype!~"tmpfs|ramfs|squashfs"})
disk_capacity: sum(node_filesystem_size_bytes{mountpoint="/",fstype!~"tmpfs|ramfs|squashfs"})
pod_count: sum(kube_pod_status_phase{phase=~"Pending|Running"})
